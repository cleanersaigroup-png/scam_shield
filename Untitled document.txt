# Translation Helper Function using `translate` library
def translate_text(text, target_lang):
    if not text or target_lang == "en":
        return text
    try:
        translator = Translator(to_lang=target_lang)
        return translator.translate(text)
    except Exception as e:
        print(f"Translation Error ({target_lang}):", e)
        return text

def normalize_phone(phone):
    if not phone:
        return ""
    digits = re.sub(r"\D", "", str(phone))
    if len(digits) >= 10:
        return digits[-10:]
    return digits




def get_reputation(phone):
    phone = normalize_phone(phone)
    if not phone:
        return {"found": False, "reports": 0, "category": "Unknown"}
    try:
        df = pd.read_csv(REPUTATION_FILE)
        if df.empty:
            return {"found": False, "reports": 0, "category": "Unknown"}
        df["phone_number"] = (
            df["phone_number"].astype(str).apply(normalize_phone)
        )
        result = df[df["phone_number"] == phone]
        if result.empty:
            return {"found": False, "reports": 0, "category": "Unknown"}
        return {
            "found": True,
            "reports": int(result.iloc[0]["reports"]),
            "category": str(result.iloc[0].get("category", "Reported")),
        }
    except Exception as e:
        print("Reputation error:", e)
        return {"found": False, "reports": 0, "category": "Unknown"}




def get_community_reports(phone):
    phone = normalize_phone(phone)
    if not phone:
        return []
    try:
        df = pd.read_csv(COMMUNITY_FILE)
        if df.empty:
            return []
        df["phone_number"] = (
            df["phone_number"].astype(str).apply(normalize_phone)
        )
        results = df[df["phone_number"] == phone]
        reports = []
        for _, row in results.iterrows():
            reports.append(
                {
                    "category": str(row.get("category", "Other")),
                    "description": str(row.get("description", "")),
                    "timestamp": str(row.get("timestamp", "")),
                }
            )
        return reports
    except Exception as e:
        print("Community database error:", e)
        return []




def add_community_report(phone, category, description):
    phone = normalize_phone(phone)
    if not phone:
        return False
    try:
        df = pd.read_csv(COMMUNITY_FILE)
        new_row = pd.DataFrame(
            [
                {
                    "phone_number": phone,
                    "category": category,
                    "description": description,
                    "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                }
            ]
        )
        df = pd.concat([df, new_row], ignore_index=True)
        df.to_csv(COMMUNITY_FILE, index=False)
        return True
    except Exception as e:
        print("Could not save report:", e)
        return False

def detect_language(text):
    if not text:
        return "Unknown"
    devanagari = len(re.findall(r"[\u0900-\u097F]", text))
    english = len(re.findall(r"[A-Za-z]", text))
    total = devanagari + english
    if total == 0:
        return "Unknown"
    if devanagari > 0 and english > 0:
        return (
            "Hindi + English" if (devanagari / total) > 0.25 else "Hinglish"
        )
    if devanagari > 0:
        return "Hindi"
    return "English / Hinglish"

def analyze_url(message):
    urls = re.findall(r"(https?://[^\s]+|www\.[^\s]+)", message.lower())
    if not urls:
        return {
            "found": False,
            "suspicious": False,
            "score": 0,
            "urls": [],
            "reasons": [],
        }
    reasons, suspicious = [], False
    for url in urls:
        clean_url = url.rstrip(".,!?)]}")
        if clean_url.startswith("http://"):
            suspicious = True
            reasons.append("The link does not use HTTPS.")
        if re.search(r"https?://\d{1,3}(\.\d{1,3}){3}", clean_url):
            suspicious = True
            reasons.append("The URL uses a raw IP address.")
        if any(
            x in clean_url
            for x in [
                "bit.ly",
                "tinyurl.com",
                "t.co",
                "goo.gl",
                "is.gd",
                "cutt.ly",
                "shorturl.at",
            ]
        ):
            suspicious = True
            reasons.append("The link uses a URL-shortening service.")
        matches = [
            w
            for w in [
                "verify",
                "login",
                "secure",
                "account",
                "kyc",
                "update",
                "refund",
                "reward",
                "claim",
                "wallet",
                "password",
            ]
            if w in clean_url
        ]
        if matches:
            suspicious = True
            reasons.append("The URL contains keywords: " + ", ".join(matches))
    score = min(90, 40 + len(reasons) * 10) if suspicious else 0
    return {
        "found": True,
        "suspicious": suspicious,
        "score": score,
        "urls": urls,
        "reasons": reasons,
    }

def model_prediction(message):
    if not MODEL_READY:
        return {"probability": 0, "prediction": "Unknown"}
    try:
        X = tfidf.transform([message])
        scam_prob = float(model.predict_proba(X)[0][1])
        return {
            "probability": round(scam_prob * 100, 2),
            "prediction": "Scam" if scam_prob >= 0.5 else "Likely Safe",
        }
    except Exception as e:
        print("Prediction error:", e)
        return {"probability": 0, "prediction": "Unknown"}

def calculate_risk(
    ai_score, url_data, reputation_data, community_count, techniques
):
    signals, weights = [ai_score], [0.70]
    if url_data["found"]:
        signals.append(url_data["score"])
        weights.append(0.15)
    if reputation_data["reports"] > 0:
        signals.append(min(90, reputation_data["reports"] * 10))
        weights.append(0.15)
    if community_count > 0:
        signals.append(community_risk_score(community_count))
        weights.append(0.15)


    if not signals:
        return 0
    risk = sum(s * w for s, w in zip(signals, weights)) / sum(weights)
    if len(techniques) >= 3:
        risk = max(risk, 70)
    if len(techniques) >= 4:
        risk = max(risk, 85)
    return round(min(100, max(0, risk)), 1)




def risk_label(score):

@app.route("/predict", methods=["POST"])
def predict():
    data = request.get_json()
    message = str(data.get("message", "")).strip()
    phone = str(data.get("phone", "")).strip()
    target_lang = str(data.get("target_lang", "en")).strip()


    if not message:
        return jsonify({"error": "Message is required."}), 400


    model_result = model_prediction(message)
    ai_score = model_result["probability"]
    category = detect_category(message)
    language = detect_language(message)
    techniques = detect_scam_techniques(message)
    url_data = analyze_url(message)
    phone_data = get_reputation(phone)
    community_reports = get_community_reports(phone)
    community_count = len(community_reports)


    risk = calculate_risk(
        ai_score, url_data, phone_data, community_count, techniques
    )
    explanations = generate_explanation(
        message, ai_score, url_data, phone_data, community_count, techniques
    )
    action_plan = generate_action_plan(category)


    if target_lang != "en":
        category = translate_text(category, target_lang)
        for tech in techniques:
            tech["type"] = translate_text(tech["type"], target_lang)
            tech["title"] = translate_text(tech["title"], target_lang)
            tech["description"] = translate_text(
                tech["description"], target_lang
            )


        for exp in explanations:
            exp["title"] = translate_text(exp["title"], target_lang)
            exp["description"] = translate_text(exp["description"], target_lang)


        action_plan = [translate_text(act, target_lang) for act in action_plan]


    return jsonify({
        "risk_score": risk,
        "risk_label": risk_label(risk),
        "ai_probability": ai_score,
        "prediction": model_result["prediction"],
        "category": category,
        "language": language,
        "techniques": techniques,
        "explanations": explanations,
        "url": url_data,
        "phone": {
            "reports": phone_data["reports"],
            "category": phone_data["category"],
            "found": phone_data["found"],
        },
        "community_reports": community_count,
        "action_plan": action_plan,
    })




@app.route("/report", methods=["POST"])
def report():
    data = request.get_json()
    phone = str(data.get("phone", "")).strip()
    category = str(data.get("category", "Other")).strip()
    description = str(data.get("description", "")).strip()


    normalized = normalize_phone(phone)
    if len(normalized) != 10:
        return jsonify(
            {"error": "Please enter a valid 10-digit phone number."}
        ), 400


    if category not in REPORT_CATEGORIES:
        category = "Other"


    if not add_community_report(normalized, category, description):
        return jsonify({"error": "Could not save report."}), 500


    return jsonify({"success": True, "message": "Report submitted successfully."})
